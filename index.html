// ============================================================================
// MOTOR EDITORIAL MIM - BANCO DE DADOS EM GOOGLE SHEETS (7 CADERNOS)
// ============================================================================
const SHEET_SECOES = "DB_SECOES_CADERNOS";
const SHEET_GLOBAL = "DB_ESTADO_GLOBAL";
const SPREADSHEET_ID_FIXO = "";

function instalarBancoDeDados() {
  const ss = getOrCreateSpreadsheet();
  ensureSheetsStructure(ss);
  Logger.log("✅ Banco pronto! URL: " + ss.getUrl());
}

function getOrCreateSpreadsheet() {
  let ss = null;
  if (SPREADSHEET_ID_FIXO && SPREADSHEET_ID_FIXO.trim() !== "") {
    try { ss = SpreadsheetApp.openById(SPREADSHEET_ID_FIXO.trim()); } catch (e) {}
  }
  if (!ss) {
    try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  }
  if (!ss) {
    const props = PropertiesService.getScriptProperties();
    const savedId = props.getProperty("MIM_DB_SPREADSHEET_ID");
    if (savedId) {
      try { ss = SpreadsheetApp.openById(savedId); } catch (e) {}
    }
    if (!ss) {
      ss = SpreadsheetApp.create("Banco_de_Dados_Matriz_MIM_7_Cadernos");
      props.setProperty("MIM_DB_SPREADSHEET_ID", ss.getId());
    }
  }
  return ss;
}

function ensureSheetsStructure(ss) {
  if (!ss) ss = getOrCreateSpreadsheet();
  let shSec = ss.getSheetByName(SHEET_SECOES);
  if (!shSec) {
    shSec = ss.insertSheet(SHEET_SECOES);
    shSec.appendRow([
      "CHAVE_UNICA", "ID_CADERNO", "NOME_CADERNO", "ID_SECAO", "TITULO_SECAO",
      "ARGUMENTOS_JSON", "TEXTO_GERADO", "MUNICIPIO_REF", "ATUALIZADO_EM"
    ]);
    shSec.getRange("A1:I1").setFontWeight("bold").setBackground("#003366").setFontColor("#ffffff");
    shSec.setFrozenRows(1);
  }
  let shGlob = ss.getSheetByName(SHEET_GLOBAL);
  if (!shGlob) {
    shGlob = ss.insertSheet(SHEET_GLOBAL);
    shGlob.appendRow(["CHAVE", "VALOR_JSON", "ATUALIZADO_EM"]);
    shGlob.getRange("A1:C1").setFontWeight("bold").setBackground("#0a192f").setFontColor("#d4af37");
    shGlob.setFrozenRows(1);
  }
  return { ss, shSec, shGlob };
}

function doGet(e) {
  try {
    const { ss, shSec, shGlob } = ensureSheetsStructure();
    const secRows = shSec.getDataRange().getValues();
    const sectionsMap = {};
    for (let i = 1; i < secRows.length; i++) {
      const [uniqueKey, cadId, cadName, secId, secTitle, argsJson, textoGerado, municipio, updatedAt] = secRows[i];
      if (!cadId || !secId) continue;
      if (!sectionsMap[cadId]) sectionsMap[cadId] = {};
      let parsedArgs = [];
      try { parsedArgs = JSON.parse(argsJson || "[]"); } catch (err) { parsedArgs = []; }
      sectionsMap[cadId][secId] = {
        title: secTitle || "",
        args: parsedArgs,
        text: textoGerado || "",
        municipio: municipio || "Tanguá",
        updatedAt: updatedAt || ""
      };
    }
    const globRows = shGlob.getDataRange().getValues();
    let anchors = null;
    let lastMunicipio = "Tanguá";
    for (let j = 1; j < globRows.length; j++) {
      if (globRows[j][0] === "ANCHORS_LIBRARY") {
        try { anchors = JSON.parse(globRows[j][1]); } catch (err) {}
      }
      if (globRows[j][0] === "ACTIVE_MUNICIPIO") {
        lastMunicipio = globRows[j][1] || "Tanguá";
      }
    }
    return ContentService.createTextOutput(JSON.stringify({
      status: "ok",
      spreadsheetUrl: ss.getUrl(),
      hasData: Object.keys(sectionsMap).length > 0,
      sectionsMap: sectionsMap,
      anchors: anchors,
      municipio: lastMunicipio,
      serverTime: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    const rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    const payload = JSON.parse(rawBody);
    const { ss, shSec, shGlob } = ensureSheetsStructure();
    const now = new Date().toISOString();

    if (payload.workbooks) {
      const existingData = shSec.getDataRange().getValues();
      const rowIndexByKey = {};
      for (let i = 1; i < existingData.length; i++) {
        rowIndexByKey[existingData[i][0]] = i + 1;
      }
      const municipio = payload.municipio || "Tanguá";
      Object.keys(payload.workbooks).forEach(function(cadId) {
        const cad = payload.workbooks[cadId];
        if (!cad || !cad.sections) return;
        Object.keys(cad.sections).forEach(function(secId) {
          const sec = cad.sections[secId];
          const uniqueKey = cadId + "__" + secId;
          const rowValues = [
            uniqueKey, cadId, cad.shortTab || cad.fullTitle || cadId, secId,
            sec.title || "", JSON.stringify(sec.args || []), sec.text || "",
            municipio, now
          ];
          if (rowIndexByKey[uniqueKey]) {
            shSec.getRange(rowIndexByKey[uniqueKey], 1, 1, rowValues.length).setValues([rowValues]);
          } else {
            shSec.appendRow(rowValues);
            rowIndexByKey[uniqueKey] = shSec.getLastRow();
          }
        });
      });
    }

    shGlob.clearContents();
    shGlob.appendRow(["CHAVE", "VALOR_JSON", "ATUALIZADO_EM"]);
    if (payload.anchors) shGlob.appendRow(["ANCHORS_LIBRARY", JSON.stringify(payload.anchors), now]);
    if (payload.municipio) shGlob.appendRow(["ACTIVE_MUNICIPIO", payload.municipio, now]);

    return ContentService.createTextOutput(JSON.stringify({
      status: "ok", spreadsheetUrl: ss.getUrl(), savedAt: now
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
