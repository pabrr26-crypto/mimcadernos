// ============================================================================
// MOTOR EDITORIAL MIM (UENF / FIPEDS) - INTEGRAÇÃO GOOGLE SHEETS (Code.gs)
// ============================================================================
// POR QUE OCORREU O ERRO "reading 'getSheetByName'"?
// 1) Se você clicar no botão "Executar" direto na barra do editor com a função
//    "ensureSheetsStructure" selecionada, o parâmetro "ss" entra vazio (undefined).
//    Para testar pelo botão Executar do editor, selecione a função "instalarBancoDeDados" abaixo!
// 2) Se o seu script foi criado de forma autônoma (fora de uma planilha), o comando
//    getActiveSpreadsheet() retorna nulo. O código abaixo corrige os DOIS casos automaticamente:
//    se não houver planilha vinculada, ele cria uma automaticamente no seu Google Drive!

const SHEET_SECOES = "DB_SECOES_CADERNOS";
const SHEET_GLOBAL = "DB_ESTADO_GLOBAL";
const SPREADSHEET_ID_FIXO = ""; // Opcional: Se quiser usar uma planilha específica, cole o ID dela aqui

// 👉 SE QUISER TESTAR PELO BOTÃO "EXECUTAR" DO APPS SCRIPT, SELECIONE ESTA FUNÇÃO:
function instalarBancoDeDados() {
  const ss = getOrCreateSpreadsheet();
  ensureSheetsStructure(ss);
  Logger.log("✅ Banco de dados MIM pronto! Planilha conectada: " + ss.getUrl());
}

function getOrCreateSpreadsheet() {
  let ss = null;
  
  // 1. Tenta pelo ID fixo, se informado
  if (SPREADSHEET_ID_FIXO && SPREADSHEET_ID_FIXO.trim() !== "") {
    try { ss = SpreadsheetApp.openById(SPREADSHEET_ID_FIXO.trim()); } catch (e) {}
  }
  
  // 2. Tenta pegar a planilha ativa (quando aberto via Extensões > Apps Script)
  if (!ss) {
    try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  }
  
  // 3. Se for um script autônomo (standalone), recupera ou cria a planilha automaticamente!
  if (!ss) {
    const props = PropertiesService.getScriptProperties();
    const savedId = props.getProperty("MIM_DB_SPREADSHEET_ID");
    if (savedId) {
      try { ss = SpreadsheetApp.openById(savedId); } catch (e) {}
    }
    if (!ss) {
      ss = SpreadsheetApp.create("Banco_de_Dados_Motor_Editorial_MIM_UENF");
      props.setProperty("MIM_DB_SPREADSHEET_ID", ss.getId());
    }
  }
  return ss;
}

function ensureSheetsStructure(ss) {
  // Blindagem contra "undefined": se "ss" não for passado, busca ou cria automaticamente
  if (!ss) {
    ss = getOrCreateSpreadsheet();
  }

  let shSec = ss.getSheetByName(SHEET_SECOES);
  if (!shSec) {
    shSec = ss.insertSheet(SHEET_SECOES);
    shSec.appendRow([
      "CHAVE_UNICA",
      "ID_CADERNO",
      "NOME_CADERNO",
      "ID_SECAO",
      "TITULO_SECAO",
      "ARGUMENTOS_JSON",
      "TEXTO_GERADO",
      "MUNICIPIO_REF",
      "ATUALIZADO_EM"
    ]);
    shSec.getRange("A1:I1").setFontWeight("bold").setBackground("#003366").setFontColor("#ffffff");
    shSec.setFrozenRows(1);
  }

  let shGlob = ss.getSheetByName(SHEET_GLOBAL);
  if (!shGlob) {
    shGlob = ss.insertSheet(SHEET_GLOBAL);
    shGlob.appendRow(["CHAVE", "VALOR_JSON", "ATUALIZADO_EM"]);
    shGlob.getRange("A1:C1").setFontWeight("bold").setBackground("#0a192f").setFontColor("#d4af37");
    shGlob.setFrozenRows(1);
  }
  return { ss, shSec, shGlob };
}

function doGet(e) {
  try {
    const { ss, shSec, shGlob } = ensureSheetsStructure();

    // 1. Lê todas as linhas da aba estruturada por seção
    const secRows = shSec.getDataRange().getValues();
    const sectionsMap = {};
    for (let i = 1; i < secRows.length; i++) {
      const [uniqueKey, cadId, cadName, secId, secTitle, argsJson, textoGerado, municipio, updatedAt] = secRows[i];
      if (!cadId || !secId) continue;
      if (!sectionsMap[cadId]) sectionsMap[cadId] = {};
      let parsedArgs = [];
      try { parsedArgs = JSON.parse(argsJson || "[]"); } catch (err) { parsedArgs = []; }
      sectionsMap[cadId][secId] = {
        title: secTitle || "",
        args: parsedArgs,
        text: textoGerado || "",
        municipio: municipio || "Tanguá",
        updatedAt: updatedAt || ""
      };
    }

    // 2. Lê acervo de âncoras e metadados globais
    const globRows = shGlob.getDataRange().getValues();
    let anchors = null;
    let lastMunicipio = "Tanguá";
    for (let j = 1; j < globRows.length; j++) {
      if (globRows[j][0] === "ANCHORS_LIBRARY") {
        try { anchors = JSON.parse(globRows[j][1]); } catch (err) {}
      }
      if (globRows[j][0] === "ACTIVE_MUNICIPIO") {
        lastMunicipio = globRows[j][1] || "Tanguá";
      }
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "ok",
      spreadsheetUrl: ss.getUrl(),
      hasData: Object.keys(sectionsMap).length > 0,
      sectionsMap: sectionsMap,
      anchors: anchors,
      municipio: lastMunicipio,
      serverTime: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    const rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    const payload = JSON.parse(rawBody);
    const { ss, shSec, shGlob } = ensureSheetsStructure();
    const now = new Date().toISOString();

    // Atualiza ou insere seções enviadas na aba DB_SECOES_CADERNOS
    if (payload.workbooks) {
      const existingData = shSec.getDataRange().getValues();
      const rowIndexByKey = {};
      for (let i = 1; i < existingData.length; i++) {
        rowIndexByKey[existingData[i][0]] = i + 1;
      }

      const municipio = payload.municipio || "Tanguá";
      Object.keys(payload.workbooks).forEach(function(cadId) {
        const cad = payload.workbooks[cadId];
        if (!cad || !cad.sections) return;
        Object.keys(cad.sections).forEach(function(secId) {
          const sec = cad.sections[secId];
          const uniqueKey = cadId + "__" + secId;
          const rowValues = [
            uniqueKey,
            cadId,
            cad.shortTab || cad.fullTitle || cadId,
            secId,
            sec.title || "",
            JSON.stringify(sec.args || []),
            sec.text || "",
            municipio,
            now
          ];

          if (rowIndexByKey[uniqueKey]) {
            shSec.getRange(rowIndexByKey[uniqueKey], 1, 1, rowValues.length).setValues([rowValues]);
          } else {
            shSec.appendRow(rowValues);
            rowIndexByKey[uniqueKey] = shSec.getLastRow();
          }
        });
      });
    }

    // Salva as âncoras de referência e município ativo na aba DB_ESTADO_GLOBAL
    shGlob.clearContents();
    shGlob.appendRow(["CHAVE", "VALOR_JSON", "ATUALIZADO_EM"]);
    if (payload.anchors) {
      shGlob.appendRow(["ANCHORS_LIBRARY", JSON.stringify(payload.anchors), now]);
    }
    if (payload.municipio) {
      shGlob.appendRow(["ACTIVE_MUNICIPIO", payload.municipio, now]);
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "ok",
      spreadsheetUrl: ss.getUrl(),
      savedAt: now
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
💡 Como funciona a tabela no Google Sheets: O script cria automaticamente a aba DB_SECOES_CADERNOS contendo uma linha para cada seção dos 7 cadernos (colunas: ID_CADERNO, ID_SECAO, TITULO_SECAO, ARGUMENTOS_JSON, TEXTO_GERADO) e a aba DB_ESTADO_GLOBAL para o acervo de referências. Assim, qualquer alteração feita na aplicação (ou diretamente nas células da planilha) fica disponível para todos que abrirem o sistema!
